## 📝 Code Patterns Established

### Loading Pattern
```tsx
// Initial page load
if (loading) return <DashboardSkeleton />;
if (error) return <ErrorDisplay onRetry={refetch} />;
if (!data) return null;

// Inline loading (like courses list)
{loading ? (
  <div className="space-y-3">
    <div className="h-20 bg-gray-100 rounded-lg animate-pulse"></div>
  </div>
) : ...}
```

### Button Loading Pattern
```tsx
<ButtonWithLoading
  onClick={handleSubmit}
  isLoading={mutation.isPending}
  className="px-4 py-2 bg-blue-600 text-white rounded-lg"
>
  Submit
</ButtonWithLoading>
```

### Validation Pattern
```tsx
if (!value) {
  const msg = 'Field is required';
  setError(msg);
  showToast.error(msg); // Show toast too!
  return;
}
```

### Error Handling Pattern
```tsx
try {
  await apiCall();
  showToast.success('Success message');
} catch (error) {
  showToast.apiError(error);
}
```

### Empty State Pattern
```tsx
if (data.length === 0) {
  return <NoDataFound onAdd={handleAdd} />;
}
```

### Selection State Pattern (NO gradients!)
```tsx
className={`border rounded-lg transition-colors ${
  isSelected
    ? 'border-blue-500 bg-blue-50'
    : 'border-gray-200 hover:border-gray-300 hover:bg-gray-50'
}`}

## 💡 How to Use Form Validation

### Example 1: Using validateFormWithToast
```typescript
import { validateFormWithToast, studentValidationRules } from '../lib/validation';

const handleSubmit = (e: React.FormEvent) => {
  e.preventDefault();
  
  if (!validateFormWithToast(formData, studentValidationRules)) {
    return; // Validation failed, toast shown automatically
  }
  
  // Proceed with submission
  mutation.mutate(formData);
};
```

### Example 2: Using useFormValidation Hook
```typescript
import { useFormValidation } from '../hooks/useFormValidation';
import { courseValidationRules } from '../lib/validation';

const { errors, validate, validateSingleField } = useFormValidation({
  rules: courseValidationRules,
});

const handleSubmit = (e: React.FormEvent) => {
  e.preventDefault();
  
  if (!validate(formData)) {
    return; // Validation failed, errors state updated
  }
  
  mutation.mutate(formData);
};

// Real-time validation on blur
const handleBlur = (e: React.FocusEvent<HTMLInputElement>) => {
  validateSingleField(e.target.name, e.target.value);
};
```

### Example 3: Using Form Components
```typescript
import { FormInput, FormSelect } from '../components/FormInput';

<FormInput
  label="Course Code"
  name="code"
  value={formData.code}
  onChange={handleChange}
  onBlur={handleBlur}
  error={errors.code}
  required
  placeholder="FA101"
/>

<FormSelect
  label="Level"
  name="level"
  value={formData.level}
  onChange={handleChange}
  options={levels.map(l => ({ value: l.id, label: l.name }))}
  error={errors.level}

/>
```
